# Real Leaderboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the mocked static leaderboard with a real Supabase-backed global leaderboard showing today's / this week's / all-time best catches, with anonymous player identity and inline name editing.

**Architecture:** React SPA calls Supabase REST directly via the anon key. Player identity is a UUID stored in `localStorage` (already generated by `getPlayerId()` in `storageManager.js`). After each cast, a fire-and-forget call upserts the player and inserts the catch. The `Leaderboard` component fetches via a Postgres stored function and caches tab results in a ref.

**Tech Stack:** Supabase (Postgres + REST + RLS), `@supabase/supabase-js`, Vite env vars (`import.meta.env`), React 18, Vitest.

  ```sql
  -- ── Tables ────────────────────────────────────────────────────
  CREATE TABLE players (
    player_id     UUID PRIMARY KEY,
    display_name  TEXT NOT NULL,
    first_seen_at TIMESTAMPTZ DEFAULT now(),
    last_seen_at  TIMESTAMPTZ DEFAULT now()
  );

  CREATE TABLE catches (
    id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    player_id     UUID NOT NULL REFERENCES players(player_id),
    fish_name     TEXT NOT NULL,
    rarity        TEXT CHECK (rarity IN ('common','uncommon','rare','epic','legendary')),
    weight_lbs    NUMERIC(4,1) CHECK (weight_lbs > 0 AND weight_lbs < 200),
    points        INTEGER CHECK (points >= 0 AND points <= 5000),
    spot_id       TEXT,
    lure_id       TEXT,
    cast_style_id TEXT,
    caught_at     TIMESTAMPTZ DEFAULT now()
  );

  -- ── Row-Level Security ────────────────────────────────────────
  ALTER TABLE players ENABLE ROW LEVEL SECURITY;
  ALTER TABLE catches ENABLE ROW LEVEL SECURITY;

  -- players: public read, anon insert + update (upgrade path tightens update)
  CREATE POLICY "players_select_public" ON players
    FOR SELECT USING (true);
  CREATE POLICY "players_insert_anon" ON players
    FOR INSERT WITH CHECK (true);
  CREATE POLICY "players_update_anon" ON players
    FOR UPDATE USING (true) WITH CHECK (true);

  -- catches: public read, anon insert (upgrade path: service key only)
  CREATE POLICY "catches_select_public" ON catches
    FOR SELECT USING (true);
  CREATE POLICY "catches_insert_anon" ON catches
    FOR INSERT WITH CHECK (true);

  -- ── Indexes ────────────────────────────────────────────────────
  CREATE INDEX idx_catches_caught_at ON catches (caught_at DESC);
  CREATE INDEX idx_catches_points    ON catches (points DESC);
  CREATE INDEX idx_catches_player_id ON catches (player_id);

  -- ── Leaderboard function ───────────────────────────────────────
  CREATE OR REPLACE FUNCTION get_leaderboard(period TEXT)
  RETURNS TABLE (
    player_id    UUID,
    display_name TEXT,
    fish_name    TEXT,
    rarity       TEXT,
    weight_lbs   NUMERIC,
    points       INTEGER,
    caught_at    TIMESTAMPTZ
  )
  LANGUAGE SQL
  STABLE
  AS $$
    WITH best AS (
      SELECT DISTINCT ON (c.player_id)
        c.player_id,
        p.display_name,
        c.fish_name,
        c.rarity,
        c.weight_lbs,
        c.points,
        c.caught_at
      FROM catches c
      JOIN players p ON c.player_id = p.player_id
      WHERE
        CASE period
          WHEN 'today' THEN c.caught_at >= CURRENT_DATE
          WHEN 'week'  THEN c.caught_at >= date_trunc('week', NOW())
          ELSE TRUE
        END
      ORDER BY c.player_id, c.points DESC
    )
    SELECT * FROM best
    ORDER BY points DESC
    LIMIT 50
  $$;
  ```

  ```js
  // src/utils/leaderboardApi.js
  // Leaderboard API: player identity, catch submission, leaderboard fetch, inline rename.
  // All Supabase errors are swallowed — this is a free-to-play game; network failures
  // must never surface to the player as errors.
  import { supabase }    from './supabase.js';
  import { getPlayerId } from './storageManager.js';

  const DISPLAY_NAME_KEY = 'brdc_display_name';

  /**
   * Returns the player's display name from localStorage.
   * If none is stored, generates "Angler_XXXX" from the last 4 chars of their UUID
   * and persists it to localStorage.
   */
  export function getDisplayName() {
    const stored = localStorage.getItem(DISPLAY_NAME_KEY);
    if (stored) return stored;

    const playerId = getPlayerId();                        // creates UUID if missing
    const suffix   = playerId.slice(-4);
    const name     = `Angler_${suffix}`;
    localStorage.setItem(DISPLAY_NAME_KEY, name);
    return name;
  }

  /**
   * Fire-and-forget: upsert the player record then insert the catch.
   * Any error (network or Supabase API) is silently swallowed.
   * localStorage has already been updated before this is called.
   */
  export async function submitCatch(result) {
    try {
      const playerId    = getPlayerId();
      const displayName = getDisplayName();
      const now         = new Date().toISOString();

      await supabase.from('players').upsert(
        { player_id: playerId, display_name: displayName, last_seen_at: now },
        { onConflict: 'player_id' },
      );

      await supabase.from('catches').insert({
        player_id:     playerId,
        fish_name:     result.fishName,
        rarity:        result.rarity,
        weight_lbs:    result.weight,
        points:        result.points,
        spot_id:       result.spot       ?? null,
        lure_id:       result.lure       ?? null,
        cast_style_id: result.castStyle  ?? null,
        caught_at:     now,
      });
    } catch {
      // Fire-and-forget: silently swallow all errors
    }
  }

  /**
   * Fetch the leaderboard for the given period.
   * period: 'today' | 'week' | 'alltime'
   * Returns an array of rows, or [] on any error.
   */
  export async function fetchLeaderboard(period) {
    try {
      const { data, error } = await supabase.rpc('get_leaderboard', { period });
      if (error) return [];
      return data ?? [];
    } catch {
      return [];
    }
  }

  /**
   * Rename the current player.
   * Validates: non-empty, max 20 chars, alphanumeric + underscores + spaces only.
   * Returns true on success, false on validation failure or Supabase error.
   * Only updates localStorage after a successful Supabase upsert.
   */
  export async function renamePlayer(displayName) {
    const trimmed = displayName.trim();
    if (!trimmed)               return false;
    if (trimmed.length > 20)    return false;
    if (!/^[\w\s]+$/.test(trimmed)) return false;   // \w = [a-zA-Z0-9_], \s = spaces

    try {
      const playerId = getPlayerId();
      const { error } = await supabase.from('players').upsert(
        { player_id: playerId, display_name: trimmed, last_seen_at: new Date().toISOString() },
        { onConflict: 'player_id' },
      );
      if (error) return false;

      localStorage.setItem(DISPLAY_NAME_KEY, trimmed);
      return true;
    } catch {
      return false;
    }
  }
  ```

  ```jsx
  import { getPlayerId } from '../utils/storageManager.js';

  const TABS = [
    { key: 'today',   label: 'Today'     },
    { key: 'week',    label: 'This Week' },
    { key: 'alltime', label: 'All-Time'  },
  ];

  // Skeleton widths for the name column (one row each)
  const SKELETON_WIDTHS = ['30%', '60%', '20%', '45%', '15%'];

  function SkeletonRow({ nameWidth }) {
    return (
      <div className="lb__row lb__row--skeleton" aria-hidden="true">
        <div className="lb__skel lb__skel--rank" />
        <div className="lb__skel lb__skel--name" style={{ width: nameWidth }} />
        <div className="lb__skel lb__skel--fish" />
        <div className="lb__skel lb__skel--pts" />
      </div>

  export default function Leaderboard({ playerResult, playerId }) {
    const [activeTab,    setActiveTab]    = useState('today');
    const [rows,         setRows]         = useState(null);   // null = not yet loaded
    const [loading,      setLoading]      = useState(false);
    const [error,        setError]        = useState(false);
    const [editingName,  setEditingName]  = useState(false);
    const [editValue,    setEditValue]    = useState('');
    const [displayName,  setDisplayName]  = useState(() => getDisplayName());

    // Cache fetched tab results so back-switching is instant
    const cacheRef = useRef({ today: null, week: null, alltime: null });

    const loadTab = useCallback(async (tab) => {
      if (cacheRef.current[tab] !== null) {
        setRows(cacheRef.current[tab]);
        return;
      }
      setLoading(true);
      setError(false);
      const data = await fetchLeaderboard(tab);
      cacheRef.current[tab] = data;
      setRows(data);
      setLoading(false);
    }, []);

    // Load initial tab on mount
    useEffect(() => { loadTab('today'); }, [loadTab]);

    function switchTab(tab) {
      setActiveTab(tab);
      loadTab(tab);
    }

    function startEdit() {
      setEditValue(displayName);
      setEditingName(true);
    }

    async function commitRename() {
      const trimmed = editValue.trim();
      setEditingName(false);
      if (!trimmed || trimmed === displayName) return;

      const ok = await renamePlayer(trimmed);
      if (ok) {
        setDisplayName(trimmed);
        // Invalidate current tab cache so next view shows updated name
        cacheRef.current[activeTab] = null;
        loadTab(activeTab);
      }
    }

    function handleKeyDown(e) {
      if (e.key === 'Enter')  commitRename();
      if (e.key === 'Escape') setEditingName(false);
    }

    // ── Render ────────────────────────────────────────────────────
    const currentPlayerId = playerId ?? getPlayerId();

    return (
      <div className="lb">
        {/* Header */}
        <div className="lb__header">
          <h3 className="lb__title">Leaderboard</h3>
        </div>

        {/* Tabs */}
        <div className="lb__tabs" role="tablist">
          {TABS.map(({ key, label }) => (
            <button
              key={key}
              role="tab"
              aria-selected={activeTab === key}
              className={`lb__tab ${activeTab === key ? 'lb__tab--active' : ''}`}
              onClick={() => switchTab(key)}
            >
              {label}
            </button>
          ))}
        </div>

        {/* List */}
        <div className="lb__list">
          {/* Loading skeleton */}
          {loading && SKELETON_WIDTHS.map((w, i) => (
            <SkeletonRow key={i} nameWidth={w} />
          ))}

          {/* Error state */}
          {!loading && error && (
            <p className="lb__empty lb__empty--error">Leaderboard unavailable right now.</p>
          )}

          {/* Empty state */}
          {!loading && !error && rows !== null && rows.length === 0 && (
            <p className="lb__empty">No casts recorded yet — be first.</p>
          )}

          {/* Rows */}
          {!loading && !error && rows !== null && rows.map((row, i) => {
            const isPlayer = row.player_id === currentPlayerId;
            const weightLbs = parseFloat(row.weight_lbs);

            return (
              <div
                key={row.player_id + i}
                className={`lb__row ${isPlayer ? 'lb__row--player' : ''}`}
              >
                {/* Rank */}
                <span className="lb__rank">#{i + 1}</span>

                {/* Name (editable for own row) */}
                <span className="lb__name-cell">
                  {isPlayer && editingName ? (
                    <input
                      className="lb__rename-input"
                      value={editValue}
                      maxLength={20}
                      autoFocus
                      onChange={e => setEditValue(e.target.value)}
                      onKeyDown={handleKeyDown}
                      onBlur={commitRename}
                      aria-label="Rename yourself"
                    />
                  ) : (
                    <>
                      <span className="lb__name">
                        {isPlayer ? displayName : row.display_name}
                      </span>
                      {isPlayer && (
                        <button
                          className="lb__pencil"
                          onClick={startEdit}
                          aria-label="Edit display name"
                        >✏️</button>
                      )}
                    </>
                  )}
                </span>

                {/* Fish + rarity + weight */}
                <span className="lb__fish">
                  {row.fish_name}
                  <span className={`rarity-badge rarity-badge--${row.rarity}`}>
                    {row.rarity}
                  </span>
                  <span className="lb__weight">{weightLbs.toFixed(1)} lbs</span>
                </span>

                {/* Points */}
                <span className="lb__pts">{row.points.toLocaleString()}</span>
              </div>
            );
          })}
        </div>
      </div>
    );
  }
  ```
